// ============================================================================
//  pe_parser.cpp — educational PE parser
//
//  Reads a Windows 64-bit PE file from disk and prints its structure:
//  DOS header, NT headers (file + optional), data directories, sections.
//
//  Standard C++ only: <iostream>, <fstream>. No Windows headers, no driver,
//  no process access. Pure static analysis of a file on disk.
//
//  Build:  cl /std:c++17 /EHsc pe_parser.cpp
//  Usage:  pe_parser.exe <path-to-pe-file>
// ============================================================================

#include <cstdint>
#include <cstring>
#include <ctime>
#include <iostream>
#include <fstream>
#include <string>
#include <vector>
#include <iomanip>

// ============================================================================
//  ON-DISK STRUCTURES (packed, little-endian — as they appear in the file)
// ============================================================================

#pragma pack(push, 1)

// --- DOS header (64 bytes). Only e_magic and e_lfanew matter to us. --------
struct DosHeader
{
    uint16_t e_magic;       // 'MZ' (0x5A4D)
    uint16_t e_cblp;        // bytes on last page
    uint16_t e_cp;          // pages in file
    uint16_t e_crlc;        // relocations
    uint16_t e_cparhdr;     // size of header in paragraphs
    uint16_t e_minalloc;
    uint16_t e_maxalloc;
    uint16_t e_ss;
    uint16_t e_sp;
    uint16_t e_csum;
    uint16_t e_ip;
    uint16_t e_cs;
    uint16_t e_lfarlc;      // file address of relocation table
    uint16_t e_ovno;        // overlay number
    uint16_t e_res[4];      // reserved
    uint16_t e_oemid;
    uint16_t e_oeminfo;
    uint16_t e_res2[10];    // reserved
    int32_t  e_lfanew;      // FILE offset of the NT headers  <-- the key field
};
static_assert(sizeof(DosHeader) == 64, "DOS header must be 64 bytes");

// --- PE file header (20 bytes) — follows the 4-byte 'PE\0\0' signature -----
struct FileHeader
{
    uint16_t Machine;              // e.g. 0x8664 = x64
    uint16_t NumberOfSections;
    uint32_t TimeDateStamp;        // link time (UTC unix time)
    uint32_t PointerToSymbolTable;
    uint32_t NumberOfSymbols;
    uint16_t SizeOfOptionalHeader;
    uint16_t Characteristics;      // e.g. 0x0022 = executable image, large-address-aware
};
static_assert(sizeof(FileHeader) == 20, "File header must be 20 bytes");

// --- Optional header, magic 0x20B = PE32+ (64-bit) --------------------------
struct OptionalHeaderPE32Plus
{
    uint16_t Magic;                // 0x20B for PE32+
    uint8_t  MajorLinkerVersion;
    uint8_t  MinorLinkerVersion;
    uint32_t SizeOfCode;
    uint32_t SizeOfInitializedData;
    uint32_t SizeOfUninitializedData;
    uint32_t AddressOfEntryPoint;  // RVA of entry point
    uint32_t BaseOfCode;           // RVA of start of code
    uint64_t ImageBase;            // preferred load address
    uint32_t SectionAlignment;     // usually 0x1000
    uint32_t FileAlignment;        // usually 0x200
    uint16_t MajorOperatingSystemVersion;
    uint16_t MinorOperatingSystemVersion;
    uint16_t MajorImageVersion;
    uint16_t MinorImageVersion;
    uint16_t MajorSubsystemVersion;
    uint16_t MinorSubsystemVersion;
    uint32_t Win32VersionValue;
    uint32_t SizeOfImage;          // total image size in memory
    uint32_t SizeOfHeaders;        // combined size of DOS+NT+section headers
    uint32_t CheckSum;
    uint16_t Subsystem;            // 2 = GUI, 3 = console
    uint16_t DllCharacteristics;   // ASLR, CFG, NX flags live here
    uint64_t SizeOfStackReserve;
    uint64_t SizeOfStackCommit;
    uint64_t SizeOfHeapReserve;
    uint64_t SizeOfHeapCommit;
    uint32_t LoaderFlags;
    uint32_t NumberOfRvaAndSizes;  // number of valid DataDirectory entries
    // DataDirectory[16] follows immediately (8 bytes each, not in this struct)
};
static_assert(sizeof(OptionalHeaderPE32Plus) == 112, "PE32+ optional header must be 112 bytes");

// --- One data directory entry (8 bytes) --------------------------------------
struct DataDirectory
{
    uint32_t VirtualAddress;   // RVA
    uint32_t Size;
};

// --- Section header (40 bytes) -----------------------------------------------
struct SectionHeader
{
    char     Name[8];               // short name, NUL-padded ("May be longer" for objects)
    uint32_t VirtualSize;           // size in memory (can exceed raw size)
    uint32_t VirtualAddress;        // RVA of section start
    uint32_t SizeOfRawData;         // size on disk (file-aligned)
    uint32_t PointerToRawData;      // file offset of section start
    uint32_t PointerToRelocations;
    uint32_t PointerToLinenumbers;
    uint16_t NumberOfRelocations;
    uint16_t NumberOfLinenumbers;
    uint32_t Characteristics;
};
static_assert(sizeof(SectionHeader) == 40, "Section header must be 40 bytes");

#pragma pack(pop)

// ============================================================================
//  SMALL HELPERS
// ============================================================================

// A PE file offset <-> RVA. Sections are laid out differently on disk than in
// memory; this walks the section table to translate.
bool RvaToFileOffset(uint32_t rva, const std::vector<SectionHeader>& sections, uint32_t& outOffset)
{
    for (const auto& s : sections)
    {
        const uint32_t vsize = s.VirtualSize ? s.VirtualSize : s.SizeOfRawData;
        if (rva >= s.VirtualAddress && rva < s.VirtualAddress + vsize)
        {
            const uint32_t delta = rva - s.VirtualAddress;
            // If the section is uninitialized (BSS-like), there is no raw data.
            if (delta >= s.SizeOfRawData)
                return false;
            outOffset = s.PointerToRawData + delta;
            return true;
        }
    }
    // Headers live before the first section; RVA == file offset there.
    if (rva < 0x1000)
    {
        outOffset = rva;
        return true;
    }
    return false;
}

std::string FourCC(uint32_t v)
{
    std::string s(4, ' ');
    s[0] = static_cast<char>(v & 0xFF);
    s[1] = static_cast<char>((v >> 8) & 0xFF);
    s[2] = static_cast<char>((v >> 16) & 0xFF);
    s[3] = static_cast<char>((v >> 24) & 0xFF);
    return s;
}

// Section characteristics decoder — teaches what the flag bits mean.
void PrintSectionCharacteristics(uint32_t ch)
{
    struct Flag { uint32_t bit; const char* name; };
    static constexpr Flag flags[] = {
        { 0x00000020, "CODE" },
        { 0x00000040, "INITIALIZED_DATA" },
        { 0x00000080, "UNINITIALIZED_DATA" },
        { 0x02000000, "DISCARDABLE" },
        { 0x04000000, "NOT_CACHED" },
        { 0x08000000, "NOT_PAGED" },
        { 0x10000000, "SHARED" },
        { 0x20000000, "EXECUTE" },
        { 0x40000000, "READ" },
        { 0x80000000, "WRITE" },
    };
    bool first = true;
    for (const auto& f : flags)
    {
        if (ch & f.bit)
        {
            std::cout << (first ? "" : "|") << f.name;
            first = false;
        }
    }
}

// ============================================================================
//  MAIN — the parse itself
// ============================================================================

int main(int argc, char* argv[])
{
    if (argc != 2)
    {
        std::cerr << "usage: " << (argc > 0 ? argv[0] : "pe_parser")
                  << " <path-to-pe-file>\n";
        return 1;
    }

    // --------------------------------------------------------------------
    // Step 0: open the file and slurp it into memory.
    // (A streaming parser could read field-by-field; slurping keeps the
    //  example focused on the *format* rather than buffer management.)
    // --------------------------------------------------------------------
    std::ifstream file(argv[1], std::ios::binary);
    if (!file)
    {
        std::cerr << "error: cannot open \"" << argv[1] << "\"\n";
        return 1;
    }

    std::vector<uint8_t> data((std::istreambuf_iterator<char>(file)),
                               std::istreambuf_iterator<char>());
    if (data.size() < sizeof(DosHeader))
    {
        std::cerr << "error: file too small to contain a DOS header\n";
        return 1;
    }

    auto read_at = [&data](size_t off, void* dst, size_t n) -> bool
    {
        if (off + n > data.size() || off + n < off)
            return false;
        std::memcpy(dst, data.data() + off, n);
        return true;
    };

    std::cout << "==============================================================\n";
    std::cout << " PE analysis: " << argv[1] << "\n";
    std::cout << " file size  : " << data.size() << " bytes\n";
    std::cout << "==============================================================\n";

    // --------------------------------------------------------------------
    // Step 1: DOS header — verify 'MZ' and find the way to the NT headers.
    // --------------------------------------------------------------------
    DosHeader dos{};
    if (!read_at(0, &dos, sizeof(dos)))
        return 1;

    std::cout << "\n[DOS Header]\n";
    if (dos.e_magic != 0x5A4D)
    {
        std::cerr << "error: not a PE file (e_magic is "
                  << std::hex << dos.e_magic << ", expected 0x5A4D 'MZ')\n";
        return 1;
    }
    std::cout << "  e_magic      = 0x5A4D ('MZ')\n";
    std::cout << "  e_lfanew     = 0x" << std::hex << dos.e_lfanew
              << "   <- file offset of NT headers\n";

    // --------------------------------------------------------------------
    // Step 2: NT signature — 'PE\0\0'.
    // --------------------------------------------------------------------
    const size_t ntOff = static_cast<size_t>(dos.e_lfanew);
    uint32_t peSignature = 0;
    if (ntOff + 4 > data.size() || !read_at(ntOff, &peSignature, 4))
    {
        std::cerr << "error: e_lfanew points outside the file\n";
        return 1;
    }
    if (peSignature != 0x00004550)   // 'PE\0\0' little-endian
    {
        std::cerr << "error: bad NT signature (expected 'PE\\0\\0')\n";
        return 1;
    }
    std::cout << "  signature    = 'PE\\0\\0'  (valid)\n";

    // --------------------------------------------------------------------
    // Step 3: file header — machine type, section count, timestamp.
    // --------------------------------------------------------------------
    FileHeader fh{};
    if (!read_at(ntOff + 4, &fh, sizeof(fh)))
        return 1;

    std::cout << "\n[File Header]\n";
    const char* machine = "unknown";
    switch (fh.Machine)
    {
        case 0x014C: machine = "i386";              break;
        case 0x8664: machine = "x64";               break;
        case 0xAA64: machine = "ARM64";             break;
        case 0x01C0: machine = "ARM Thumb-2";       break;
    }
    std::cout << "  Machine              = 0x" << std::hex << fh.Machine
              << " (" << machine << ")\n";
    std::cout << "  NumberOfSections     = " << std::dec << fh.NumberOfSections << "\n";
    {
        // TimeDateStamp is a plain unix timestamp in newer toolchains
        const std::time_t ts = static_cast<std::time_t>(fh.TimeDateStamp);
        const std::tm*    tm = std::gmtime(&ts);
        char buf[32]{};
        if (tm && std::strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S UTC", tm))
            std::cout << "  TimeDateStamp        = 0x" << std::hex << fh.TimeDateStamp
                      << "  (" << buf << ")\n";
        else
            std::cout << "  TimeDateStamp        = 0x" << std::hex << fh.TimeDateStamp << "\n";
    }
    std::cout << "  SizeOfOptionalHeader = 0x" << std::hex << fh.SizeOfOptionalHeader << "\n";
    std::cout << "  Characteristics      = 0x" << std::hex << fh.Characteristics << "\n";

    // --------------------------------------------------------------------
    // Step 4: optional header — we only fully decode PE32+ (64-bit images).
    // --------------------------------------------------------------------
    const size_t optOff = ntOff + 4 + sizeof(FileHeader);
    if (optOff + 2 > data.size())
    {
        std::cerr << "error: truncated optional header\n";
        return 1;
    }

    uint16_t optMagic = 0;
    std::memcpy(&optMagic, data.data() + optOff, 2);
    if (optMagic != 0x20B)
    {
        std::cerr << "note: this image is PE32 (0x" << std::hex << optMagic
                  << "); this tool demonstrates the PE32+ layout only.\n";
        return 1;
    }

    OptionalHeaderPE32Plus opt{};
    if (!read_at(optOff, &opt, sizeof(opt)))
        return 1;

    std::cout << "\n[Optional Header (PE32+)]\n";
    std::cout << "  Magic                = 0x20B (PE32+)\n";
    std::cout << "  LinkerVersion        = " << std::dec
              << static_cast<int>(opt.MajorLinkerVersion) << "."
              << static_cast<int>(opt.MinorLinkerVersion) << "\n";
    std::cout << "  AddressOfEntryPoint  = 0x" << std::hex << opt.AddressOfEntryPoint << "  (RVA)\n";
    std::cout << "  ImageBase            = 0x" << std::hex << opt.ImageBase << "\n";
    std::cout << "  SectionAlignment     = 0x" << std::hex << opt.SectionAlignment << "\n";
    std::cout << "  FileAlignment        = 0x" << std::hex << opt.FileAlignment << "\n";
    std::cout << "  SizeOfImage          = 0x" << std::hex << opt.SizeOfImage << "\n";
    std::cout << "  SizeOfHeaders        = 0x" << std::hex << opt.SizeOfHeaders << "\n";
    std::cout << "  Subsystem            = " << std::dec << opt.Subsystem
              << (opt.Subsystem == 2 ? "  (GUI)" :
                  opt.Subsystem == 3 ? "  (console)" : "") << "\n";

    // DllCharacteristics — where the binary "mitigation" flags live.
    std::cout << "  DllCharacteristics   = 0x" << std::hex << opt.DllCharacteristics << "  [";
    {
        bool first = true;
        auto flag = [&](uint16_t bit, const char* name)
        {
            if (opt.DllCharacteristics & bit)
            {
                std::cout << (first ? "" : ", ") << name;
                first = false;
            }
        };
        flag(0x0020, "HIGH_ENTROPY_VA");
        flag(0x0040, "DYNAMIC_BASE (ASLR)");
        flag(0x0100, "NX_COMPAT (DEP)");
        flag(0x0400, "NO_SEH");
        flag(0x8000, "TERMINAL_SERVER_AWARE");
    }
    std::cout << "]\n";

    // --------------------------------------------------------------------
    // Step 5: data directories — a table of RVAs to important structures.
    // --------------------------------------------------------------------
    const size_t ddOff = optOff + sizeof(OptionalHeaderPE32Plus);
    const uint16_t ddCount = static_cast<uint16_t>(
        (opt.NumberOfRvaAndSizes < 16) ? opt.NumberOfRvaAndSizes : 16);

    struct DdName { const char* name; };
    static constexpr DdName ddNames[16] = {
        { "Export" },            { "Import" },
        { "Resource" },          { "Exception" },
        { "Security" },          { "BaseReloc" },
        { "Debug" },             { "Architecture" },
        { "GlobalPtr" },         { "TLS" },
        { "LoadConfig" },        { "BoundImport" },
        { "IAT" },               { "DelayImport" },
        { "CLR Runtime" },       { "Reserved" },
    };

    std::cout << "\n[Data Directories]  (NumberOfRvaAndSizes = "
              << std::dec << opt.NumberOfRvaAndSizes << ")\n";
    std::vector<DataDirectory> dirs(ddCount);
    for (uint16_t i = 0; i < ddCount; i++)
    {
        if (!read_at(ddOff + i * sizeof(DataDirectory), &dirs[i], sizeof(DataDirectory)))
            break;
        if (dirs[i].VirtualAddress == 0 && dirs[i].Size == 0)
            continue;
        std::cout << "  [" << std::dec << std::setw(2) << i << "] "
                  << std::left << std::setw(14) << ddNames[i].name
                  << std::right << " RVA=0x" << std::hex << dirs[i].VirtualAddress
                  << "  Size=0x" << dirs[i].Size << "\n";
    }

    // --------------------------------------------------------------------
    // Step 6: section table — 40-byte entries right after the optional header.
    // --------------------------------------------------------------------
    const size_t secOff = optOff + fh.SizeOfOptionalHeader;
    std::vector<SectionHeader> sections(fh.NumberOfSections);

    std::cout << "\n[Section Table]\n";
    std::cout << "  #  name      VirtAddr  VirtSize  RawPtr    RawSize   flags\n";
    for (uint16_t i = 0; i < fh.NumberOfSections; i++)
    {
        if (!read_at(secOff + i * sizeof(SectionHeader), &sections[i], sizeof(SectionHeader)))
        {
            std::cerr << "error: section table truncated\n";
            return 1;
        }
        std::cout << "  " << std::dec << std::setw(2) << i << " "
                  << std::left << std::setw(10)
                  << std::string(sections[i].Name, strnlen(sections[i].Name, 8))
                  << std::right << "0x" << std::hex << std::setfill('0') << std::setw(8)
                  << sections[i].VirtualAddress << "  0x" << std::setw(8) << sections[i].VirtualSize
                  << "  0x" << std::setw(8) << sections[i].PointerToRawData
                  << "  0x" << std::setw(8) << sections[i].SizeOfRawData
                  << std::setfill(' ') << "  ";
        PrintSectionCharacteristics(sections[i].Characteristics);
        std::cout << "\n";
    }

    // --------------------------------------------------------------------
    // Step 7 (bonus): locate the import directory's first descriptor on disk,
    // to show how RVAs map back to file offsets via the section table.
    // --------------------------------------------------------------------
    if (ddCount > 1 && dirs[1].VirtualAddress != 0)
    {
        uint32_t importFileOff = 0;
        std::cout << "\n[Import Directory Demo]\n";
        if (RvaToFileOffset(dirs[1].VirtualAddress, sections, importFileOff))
        {
            std::cout << "  Import dir RVA 0x" << std::hex << dirs[1].VirtualAddress
                      << " -> file offset 0x" << importFileOff << "\n";
        }
        else
        {
            std::cout << "  (could not map import RVA to a file offset)\n";
        }
    }

    std::cout << "\n==============================================================\n";
    std::cout << " parse complete\n";
    std::cout << "==============================================================\n";
    return 0;
}
Quick orientation to the three pieces you asked for:

Packed structs (lines ~30–120): each one is followed by a static_assert pinning its size to the PE spec value (64/20/112/40 bytes), so any accidental padding or field-width mistake fails at compile time rather than silently corrupting the parse. DataDirectory is the one struct without an assert only because it's a trivial 8-byte pair.
RvaToFileOffset: the translation works by walking the section table to find which section's virtual range (VirtualAddress .. VirtualAddress + VirtualSize) contains the RVA, then applying the same delta to the section's disk location (PointerToRawData). The two guard cases matter in practice: uninitialized sections (BSS-style, SizeOfRawData == 0) have no bytes on disk, and the header region before the first section is identity-mapped (RVA == file offset).
main(): takes exactly one argument, reads the whole file into a std::vector<uint8_t> via std::ifstream + istreambuf_iterator, then uses one bounds-checked read_at lambda for every struct read — that's the whole safety story. The parse order is the chain I described earlier: MZ → e_lfanew → PE\0\0 → file header → optional header → data directories → section table, with the import-directory demo at the end exercising the RVA→offset conversion for real.
Build it standalone with cl /std:c++17 /EHsc pe_parser.cpp or use the pe_parser.vcxproj already in that folder. Try it on a few different binaries — a system DLL, one of your own builds — and watch how e_lfanew varies, how ImageBase differs between EXEs (0x140000000) and DLLs (0x180000000), and which data directories each image populates. That variation is the fastest way to internalize what's fixed by the spec versus chosen by the linker.
