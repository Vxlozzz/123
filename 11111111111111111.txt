// ============================================================================
// PE PARSER - MINIMAL HEADERS
// ============================================================================

struct PESection
{
    std::string Name;
    uint8_t*    Data;
    size_t      Size;
    uint32_t    VirtualAddress;
    uint32_t    RawOffset;
    uint32_t    Characteristics;
};

struct PEImage
{
    uint8_t*               Base;
    size_t                 Size;
    uint32_t               ImageBase;
    std::vector<PESection> Sections;
    PESection*             TextSection;
    PESection*             DataSection;
    PESection*             RDataSection;
};

#pragma pack(push, 1)
struct DOSHeader
{
    uint16_t e_magic;
    uint16_t e_cblp, e_cp, e_crlc, e_cparhdr, e_minalloc, e_maxalloc;
    uint16_t e_ss, e_sp, e_csum, e_ip, e_cs, e_lfarlc, e_ovno;
    uint16_t e_res[4], e_oemid, e_oeminfo, e_res2[10];
    int32_t  e_lfanew;
};

struct FileHeader
{
    uint16_t Machine, NumberOfSections;
    uint32_t TimeDateStamp, PointerToSymbolTable, NumberOfSymbols;
    uint16_t SizeOfOptionalHeader, Characteristics;
};

struct OptionalHeader64
{
    uint16_t Magic;
    uint8_t  MajorLinkerVersion, MinorLinkerVersion;
    uint32_t SizeOfCode, SizeOfInitializedData, SizeOfUninitializedData;
    uint32_t AddressOfEntryPoint, BaseOfCode;
    uint64_t ImageBase;
    uint32_t SectionAlignment, FileAlignment;
    uint16_t MajorOperatingSystemVersion, MinorOperatingSystemVersion;
    uint16_t MajorImageVersion, MinorImageVersion;
    uint16_t MajorSubsystemVersion, MinorSubsystemVersion;
    uint32_t Win32VersionValue, SizeOfImage, SizeOfHeaders, CheckSum;
    uint16_t Subsystem, DllCharacteristics;
    uint64_t SizeOfStackReserve, SizeOfStackCommit, SizeOfHeapReserve, SizeOfHeapCommit;
    uint32_t LoaderFlags, NumberOfRvaAndSizes;
};

struct SectionHeader
{
    char     Name[8];
    uint32_t VirtualSize, VirtualAddress, SizeOfRawData, PointerToRawData;
    uint32_t PointerToRelocations, PointerToLinenumbers;
    uint16_t NumberOfRelocations, NumberOfLinenumbers;
    uint32_t Characteristics;
};
#pragma pack(pop)

bool ParsePEImage(uint8_t* Buffer, size_t BufferSize, PEImage& Image)
{
    if (BufferSize < sizeof(DOSHeader))
        return false;
    
    DOSHeader* dos = reinterpret_cast<DOSHeader*>(Buffer);
    if (dos->e_magic != 0x5A4D)
        return false;
    
    int32_t peOffset = dos->e_lfanew;
    if (peOffset < 0 || peOffset + 4 + sizeof(FileHeader) > BufferSize)
        return false;
    
    uint32_t* peSig = reinterpret_cast<uint32_t*>(Buffer + peOffset);
    if (*peSig != 0x00004550)
        return false;
    
    FileHeader* file = reinterpret_cast<FileHeader*>(Buffer + peOffset + 4);
    OptionalHeader64* opt = reinterpret_cast<OptionalHeader64*>(
        Buffer + peOffset + 4 + sizeof(FileHeader));
    
    if (opt->Magic != 0x20b)
        return false;
    
    Image.ImageBase = static_cast<uint32_t>(opt->ImageBase);
    
    SectionHeader* sections = reinterpret_cast<SectionHeader*>(
        Buffer + peOffset + 4 + sizeof(FileHeader) + file->SizeOfOptionalHeader);
    
    for (uint16_t i = 0; i < file->NumberOfSections; i++)
    {
        SectionHeader* sh = &sections[i];
        PESection section;
        section.Name = std::string(sh->Name, strnlen(sh->Name, 8));
        section.VirtualAddress = sh->VirtualAddress;
        section.RawOffset = sh->PointerToRawData;
        section.Size = sh->SizeOfRawData;
        section.Characteristics = sh->Characteristics;
        section.Data = Buffer + sh->PointerToRawData;
        Image.Sections.push_back(section);
        
        if (section.Name == ".text")
            Image.TextSection = &Image.Sections.back();
        else if (section.Name == ".data")
            Image.DataSection = &Image.Sections.back();
        else if (section.Name == ".rdata")
            Image.RDataSection = &Image.Sections.back();
    }
    
    return true;
}

// STRING LOCATOR

struct StringLocation
{
    uint32_t FileOffset;
    uint32_t RVA;
    bool     Found;
    
    StringLocation() : FileOffset(0), RVA(0), Found(false) {}
};

StringLocation FindStringInSection(
    uint8_t*    SectionData,
    size_t      SectionSize,
    uint32_t    SectionRVA,
    const char* TargetString
)
{
    StringLocation result;
    size_t len = strlen(TargetString);
    
    if (len == 0 || !SectionData)
        return result;
    
    for (size_t i = 0; i <= SectionSize - len; i++)
    {
        bool match = true;
        for (size_t j = 0; j < len; j++)
        {
            if (SectionData[i + j] != static_cast<uint8_t>(TargetString[j]))
            {
                match = false;
                break;
            }
        }
        
        if (match)
        {
            result.FileOffset = static_cast<uint32_t>(i);
            result.RVA = SectionRVA + static_cast<uint32_t>(i);
            result.Found = true;
            return result;
        }
    }
    
    return result;
}

StringLocation FindStringInImage(PEImage& Image, const char* TargetString)
{
    StringLocation result;
    
    // Check .rdata first (strings typically here)
    if (Image.RDataSection)
    {
        result = FindStringInSection(
            Image.RDataSection->Data,
            Image.RDataSection->Size,
            Image.RDataSection->VirtualAddress,
            TargetString);
        if (result.Found)
            return result;
    }
    
    // Fallback to .data
    if (Image.DataSection)
    {
        result = FindStringInSection(
            Image.DataSection->Data,
            Image.DataSection->Size,
            Image.DataSection->VirtualAddress,
            TargetString);
        if (result.Found)
            return result;
    }
    
    // Last resort: entire binary
    result = FindStringInSection(Image.Base, Image.Size, 0, TargetString);
    return result;
}

// ============================================================================
// REGISTER-INDEPENDENT XREF FINDER
// ============================================================================

struct XRefLocation
{
    uint32_t FileOffset;        // Where the displacement appears in file
    uint32_t RVA;               // Instruction RVA
    int32_t  Displacement;      // The 4-byte signed displacement
    uint8_t  InstructionSize;   // Assumed instruction size (5-7 bytes)
    bool     Found;
    
    XRefLocation() : FileOffset(0), RVA(0), Displacement(0), InstructionSize(7), Found(false) {}
};

/*
Register-Independent XREF Discovery:

Instead of matching instruction opcodes, we use pure math:
  1. For every 4-byte aligned position in .text
  2. Read the 4 bytes as a signed 32-bit displacement
  3. Check if: (CurrentAddress + InstructionSize + Displacement) == TargetRVA
  4. If true, we found an xref regardless of what instruction it belongs to

This catches ALL RIP-relative references:
  - LEA with any register (RCX, RDX, R8, R9, R11, etc.)
  - MOV with any register
  - CMP, TEST, ADD, SUB with RIP-relative operands
  - Any future instruction encodings the compiler invents

We try multiple instruction sizes (5, 6, 7 bytes) since the exact
instruction length affects the RIP-relative calculation.
*/

std::vector<XRefLocation> FindAllXRefsToRVA(
    uint8_t* CodeSection,
    size_t   CodeSize,
    uint32_t CodeSectionRVA,
    uint32_t TargetRVA
)
{
    std::vector<XRefLocation> results;
    
    if (!CodeSection || CodeSize < 4)
        return results;
    
    // Scan every byte position (not just 4-byte aligned - compilers can emit unaligned)
    for (size_t i = 0; i <= CodeSize - 4; i++)
    {
        // Read 4 bytes as potential displacement
        int32_t disp = *reinterpret_cast<int32_t*>(CodeSection + i);
        
        // Try different instruction sizes (5, 6, 7 bytes are common for RIP-relative)
        // The displacement is typically the LAST 4 bytes of the instruction
        // So we check if: (InstructionRVA + InstSize + disp) == TargetRVA
        
        for (uint8_t instSize = 5; instSize <= 7; instSize++)
        {
            // For displacement at position i within the instruction
            // The instruction started at (i - (instSize - 4))
            // But we're scanning for displacement position, so:
            // Target = RVA_of_disp + 4 + disp (RIP points to next instruction after disp)
            // Actually: RIP at execution = InstructionAddr + InstSize
            // So: Target = (CodeSectionRVA + i - (instSize - 4)) + instSize + disp
            // Simplifies to: Target = CodeSectionRVA + i + 4 + disp
            
            // Wait, let me recalculate:
            // If displacement is at byte offset i, and it's part of a 5/6/7 byte instruction
            // The instruction starts at different positions depending on size
            // But RIP-relative addressing always uses: Target = NextInstAddr + disp
            // Where NextInstAddr = InstAddr + InstSize
            
            // If disp is at position i, and we assume it's the last 4 bytes of instruction:
            // For 5-byte instruction: inst starts at i-1, RIP after = i+4, target = i+4+disp
            // For 6-byte instruction: inst starts at i-2, RIP after = i+4, target = i+4+disp
            // For 7-byte instruction: inst starts at i-3, RIP after = i+4, target = i+4+disp
            
            // All cases: target = CodeSectionRVA + i + 4 + disp
            
            uint32_t dispRVA = CodeSectionRVA + static_cast<uint32_t>(i);
            int64_t potentialTarget = static_cast<int64_t>(dispRVA) + 4 + disp;
            
            if (potentialTarget == static_cast<int64_t>(TargetRVA))
            {
                XRefLocation xref;
                
                // Calculate instruction RVA (instruction starts before displacement)
                // For a 7-byte instruction with displacement in last 4 bytes:
                //   inst start = i - 3
                // For 6-byte: inst start = i - 2
                // For 5-byte: inst start = i - 1
                
                int32_t instStart = static_cast<int32_t>(i) - (instSize - 4);
                
                if (instStart >= 0)
                {
                    xref.FileOffset = static_cast<uint32_t>(instStart);
                    xref.RVA = CodeSectionRVA + static_cast<uint32_t>(instStart);
                    xref.Displacement = disp;
                    xref.InstructionSize = instSize;
                    xref.Found = true;
                    results.push_back(xref);
                }
            }
        }
    }
    
    return results;
}

// ============================================================================
// GLOBAL OFFSET EXTRACTOR
// ============================================================================

struct GlobalOffset
{
    std::string Marker;
    uint32_t    GlobalRVA;
    int32_t     Displacement;
    bool        Found;
    
    GlobalOffset() : GlobalRVA(0), Displacement(0), Found(false) {}
};

/*
Find Global Near XRef:

Scan 0x20 bytes forward and backward from a string xref to find another
RIP-relative instruction. That instruction likely points to the global
(FNamePool or GUObjectArray).

Uses the same register-independent math as the xref finder.
*/

GlobalOffset FindGlobalNearXRef(
    PEImage&      Image,
    XRefLocation& StringXRef,
    uint32_t      CodeSectionRVA,
    const char*   MarkerName
)
{
    GlobalOffset result;
    result.Marker = MarkerName;
    
    if (!Image.TextSection || !StringXRef.Found)
        return result;
    
    uint8_t* code = Image.TextSection->Data;
    size_t codeSize = Image.TextSection->Size;
    
    size_t xrefPos = StringXRef.FileOffset;
    constexpr int32_t scanRange = 0x20;
    
    // Scan backward then forward
    for (int direction = -1; direction <= 1; direction += 2)
    {
        for (int32_t offset = 7; offset <= scanRange; offset++)
        {
            size_t scanPos;
            if (direction == -1)
                scanPos = xrefPos - offset;
            else
                scanPos = xrefPos + offset;
            
            if (scanPos < 4 || scanPos >= codeSize - 4)
                continue;
            
            // Skip if this is the string xref itself
            if (scanPos == xrefPos)
                continue;
            
            // Read potential displacement at this position
            int32_t disp = *reinterpret_cast<int32_t*>(code + scanPos);
            
            // Try instruction sizes 5-7
            for (uint8_t instSize = 5; instSize <= 7; instSize++)
            {
                uint32_t dispRVA = CodeSectionRVA + static_cast<uint32_t>(scanPos);
                int64_t targetRVA = static_cast<int64_t>(dispRVA) + 4 + disp;
                
                // Check if this displacement points somewhere in .data or .bss (likely a global)
                // For now, accept any valid displacement that's within the image bounds
                
                int32_t instStart = static_cast<int32_t>(scanPos) - (instSize - 4);
                
                if (instStart >= 0 && instStart < static_cast<int32_t>(codeSize))
                {
                    // Basic sanity: displacement should be reasonable (not pointing outside image)
                    // and not pointing to the same string we're already tracking
                    
                    if (targetRVA != static_cast<int64_t>(StringXRef.RVA + StringXRef.InstructionSize + 
                        *reinterpret_cast<int32_t*>(code + xrefPos + (StringXRef.InstructionSize - 4))))
                    {
                        result.GlobalRVA = static_cast<uint32_t>(targetRVA);
                        result.Displacement = disp;
                        result.Found = true;
                        return result;
                    }
                }
            }
        }
    }
    
    return result;
}

// ============================================================================

void ScanForFNamePool(PEImage& Image, GlobalOffset& Result)
{
    std::cout << "\n[*] Scanning for FNamePool via string references..." << std::endl;
    
    if (!Image.TextSection)
    {
        std::cout << "[-] No .text section found" << std::endl;
        return;
    }
    
    for (const char* marker : TargetStrings::FNamePoolMarkers)
    {
        std::cout << "    Trying string: \"" << marker << "\"" << std::endl;
        
        // Step 1: Find string
        StringLocation strLoc = FindStringInImage(Image, marker);
        if (!strLoc.Found)
        {
            std::cout << "    [-] String not found" << std::endl;
            continue;
        }
        
        std::cout << "    [+] String found at RVA: 0x" << std::hex << strLoc.RVA << std::endl;
        
        // Step 2: Find all xrefs (register-independent)
        std::vector<XRefLocation> xrefs = FindAllXRefsToRVA(
            Image.TextSection->Data,
            Image.TextSection->Size,
            Image.TextSection->VirtualAddress,
            strLoc.RVA);
        
        if (xrefs.empty())
        {
            std::cout << "    [-] No xrefs found" << std::endl;
            continue;
        }
        
        std::cout << "    [+] Found " << std::dec << xrefs.size() << " xref(s)" << std::endl;
        
        // Step 3: Try each xref
        for (const auto& xref : xrefs)
        {
            std::cout << "        XRef at RVA: 0x" << std::hex << xref.RVA
                      << " (disp: 0x" << xref.Displacement << ")" << std::endl;
            
            GlobalOffset global = FindGlobalNearXRef(
                Image,
                const_cast<XRefLocation&>(xref),
                Image.TextSection->VirtualAddress,
                marker);
            
            if (global.Found)
            {
                Result = global;
                Result.Marker = marker;
                std::cout << "    [+] FNamePool found at RVA: 0x"
                          << std::hex << global.GlobalRVA << std::endl;
                return;
            }
        }
    }
    
    std::cout << "[-] FNamePool not found" << std::endl;
}

{

struct FileBuffer
{
    uint8_t* Data;
    size_t   Size;
    
    FileBuffer() : Data(nullptr), Size(0) {}
    
    ~FileBuffer()
    {
        if (Data)
        {
            delete[] Data;
            Data = nullptr;
        }
    }
    
    bool Load(const char* FilePath)
    {
        std::ifstream file(FilePath, std::ios::binary | std::ios::ate);
        
        if (!file.is_open())
        {
            std::cerr << "[!] Failed to open: " << FilePath << std::endl;
            return false;
        }
        
        Size = static_cast<size_t>(file.tellg());
        file.seekg(0, std::ios::beg);
        
        Data = new uint8_t[Size];
        
        if (!file.read(reinterpret_cast<char*>(Data), Size))
        {
            std::cerr << "[!] Failed to read file" << std::endl;
            delete[] Data;
            Data = nullptr;
            Size = 0;
            return false;
        }
        
        std::cout << "[+] Loaded " << Size << " bytes from: " << FilePath << std::endl;
        return true;
    }
};

{

        targetFile = argv[1];

    {

        std::cout << "Using default: " << targetFile << std::endl;
    }
    
    std::cout << "\n[*] Loading executable..." << std::endl;
    
    FileBuffer buffer;
    if (!buffer.Load(targetFile))
    {
        std::cout << "\n[!] Ensure file is in current directory or pass full path" << std::endl;
        std::cin.get();
        return 1;
    }
    
    std::cout << "[+] File loaded" << std::endl;
    
    // Parse PE
    std::cout << "\n[*] Parsing PE image..." << std::endl;
    
    PEImage image;
    image.Base = buffer.Data;
    image.Size = buffer.Size;
    image.TextSection = nullptr;
    image.DataSection = nullptr;
    image.RDataSection = nullptr;
    
    if (!ParsePEImage(buffer.Data, buffer.Size, image))
    {
        std::cerr << "[!] Failed to parse PE" << std::endl;
        std::cin.get();
        return 1;
    }
    
    std::cout << "[+] Image base: 0x" << std::hex << image.ImageBase << std::endl;
    std::cout << "[+] Sections: " << std::dec << image.Sections.size() << std::endl;
    
    if (image.TextSection)
        std::cout << "    .text:  RVA=0x" << std::hex << image.TextSection->VirtualAddress
                  << ", Size=0x" << image.TextSection->Size << std::endl;
    if (image.RDataSection)
        std::cout << "    .rdata: RVA=0x" << std::hex << image.RDataSection->VirtualAddress
                  << ", Size=0x" << image.RDataSection->Size << std::endl;
    if (image.DataSection)
        std::cout << "    .data:  RVA=0x" << std::hex << image.DataSection->VirtualAddress
                  << ", Size=0x" << image.DataSection->Size << std::endl;
    
    // Scan for offsets
    GlobalOffset fNamePool, gUObjectArray;
    ScanForFNamePool(image, fNamePool);
    ScanForGUObjectArray(image, gUObjectArray);
    
    // Print results
    std::cout << "\n" << std::string(60, '=') << std::endl;
    std::cout << "RESULTS" << std::endl;
    std::cout << std::string(60, '=') << std::endl;
    
    if (fNamePool.Found)
    {
        std::cout << "\n[FNamePool]" << std::endl;
        std::cout << "  Marker: \"" << fNamePool.Marker << "\"" << std::endl;
        std::cout << "  RVA: 0x" << std::hex << fNamePool.GlobalRVA << std::endl;
        std::cout << "  VA:  0x" << std::hex << (image.ImageBase + fNamePool.GlobalRVA) << std::endl;
    }
    
    if (gUObjectArray.Found)
    {
        std::cout << "\n[GUObjectArray]" << std::endl;
        std::cout << "  Marker: \"" << gUObjectArray.Marker << "\"" << std::endl;
        std::cout << "  RVA: 0x" << std::hex << gUObjectArray.GlobalRVA << std::endl;
        std::cout << "  VA:  0x" << std::hex << (image.ImageBase + gUObjectArray.GlobalRVA) << std::endl;
    }
    
    if (fNamePool.Found && gUObjectArray.Found)
    {
        std::cout << "\n" << std::string(60, '-') << std::endl;
        std::cout << "COPY-PASTE FOR YOUR CODE:" << std::endl;
        std::cout << std::string(60, '-') << std::endl;
        std::cout << "\nnamespace Offsets" << std::endl;
        std::cout << "{" << std::endl;
        std::cout << "    constexpr uintptr_t FNamePoolOffset   = 0x"
                  << std::hex << fNamePool.GlobalRVA << ";" << std::endl;
        std::cout << "    constexpr uintptr_t GUObjectArrayOffset = 0x"
                  << std::hex << gUObjectArray.GlobalRVA << ";" << std::endl;
        std::cout << "}" << std::endl;
    }
    
    std::cout << "\n" << std::string(60, '=') << std::endl;
    std::cout << "\n[*] Done. Press Enter to exit..." << std::endl;
    std::cin.get();
    
    return (fNamePool.Found && gUObjectArray.Found) ? 0 : 1;
}
